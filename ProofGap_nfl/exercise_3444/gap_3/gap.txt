ASSUM:
1. a ∈ RealSet
2. b ∈ RealSet
3. A ∈ RealSet
4. y : RealSet → RealSet
5. u : RealSet → RealSet
6. t : RealSet → RealSet

10. forall (x), x ∈ RealSet ∧ x ≠ a ∧ x ≠ b ⇒ t(x) = ln(|x - a|) - ln(|x - b|)
11. forall (x), x ∈ RealSet ∧ x ≠ a ∧ x ≠ b ⇒ FunDeri(t, 1, 1)(x) = frac(1, x - a) - frac(1, x - b)

forall (x), x ∈ RealSet ∧ x ≠ a ∧ x ≠ b ⇒ frac(1, x - a) - frac(1, x - b) = frac(a - b, (x - a) * (x - b))
