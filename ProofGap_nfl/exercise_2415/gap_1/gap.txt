PROOF GAP @1
ASSUM:
1. x : RealSet → RealSet
2. y : RealSet → RealSet
3. a ∈ RealSet ∧ a > 0
4. S ∈ RealSet
5. A ∈ CartesianProd(RealSet, RealSet)
6. B ∈ CartesianProd(RealSet, RealSet)
7. Γ ⊆ CartesianProd(RealSet, RealSet)
8. AB ⊆ CartesianProd(RealSet, RealSet)
9. forall (t), t ∈ RealSet ∧ 0 ≤ t ∧ t ≤ 2 * π ⇒ x(t) = a * (cos(t) + t * sin(t))
10. forall (t), t ∈ RealSet ∧ 0 ≤ t ∧ t ≤ 2 * π ⇒ y(t) = a * (sin(t) - t * cos(t))
11. A = (a, -2 * π * a)
12. B = (a, 0)
13. Γ = { (x(t), y(t)) | t ∈ RealSet, 0 ≤ t, t ≤ 2 * π }
14. AB = { (u, v) | u ∈ RealSet, v ∈ RealSet, u = a, -2 * π * a ≤ v, v ≤ 0 }
15. S = -VectorCurveInt(Γ, (fun u, v [u ∈ RealSet ∧ v ∈ RealSet] . v) * diff(fun u, v [u ∈ RealSet ∧ v ∈ RealSet] . u)) - VectorCurveInt(AB, (fun u, v [u ∈ RealSet ∧ v ∈ RealSet] . v) * diff(fun u, v [u ∈ RealSet ∧ v ∈ RealSet] . u))

GOAL:
forall (t), t ∈ RealSet ∧ 0 < t ∧ t < 2 * π ⇒ diff(x, t) = a * t * cos(t) * diff(fun t [t ∈ RealSet] . t)
