PROOF GAP @2
ASSUM:
1. z : CartesianProd(RealSet, RealSet) → RealSet
2. h : RealSet → RealSet
3. α ∈ RealSet
4. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ z(x, y) = x^{2} - x * y + y^{2}
5. forall (t), t ∈ RealSet ⇒ h(t) = z(1 + t * cos(α), 1 + t * sin(α))
6. 0 ≤ α
7. α < 2 * π
8. FunDeri(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . z(x, y), 1, 1)(1, 1) = 1

FunDeri(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . z(x, y), 2, 1)(1, 1) = 1
