4. sum_{ n = 1 }^{ +∞ } ((1 + frac(1, n))^{-n^{2}} * e^{-(n * x)}) = sum_{ n = 1 }^{ +∞ } (a(n) * (e^{-x})^{n})
5. seqlim_{ n → +∞ } (sqrtn(n, |a(n)|)) = seqlim_{ n → +∞ } ((1 + frac(1, n))^{-n})

7. seqlim_{ n → +∞ } (sqrtn(n, |a(n)|)) = e^{-1}

13. x = -1 ⇒ seqlim_{ n → +∞ } ((1 + frac(1, n))^{-n^{2}} * e^{n}) = 1
