ASSUM:
1. u : CartesianProd(RealSet, RealSet) → RealSet
2. t : CartesianProd(RealSet, RealSet) → RealSet
3. f : RealSet → RealSet
4. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ t(x, y) = x + y

6. FuncOfClassK(f, 2)

forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ diff(u) = FunDeri(f, 1, 1)(t(x, y)) * (diff(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . x) + diff(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . y))
