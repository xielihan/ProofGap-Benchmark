ASSUM:
1. x ∈ RealSet ∧ x ∈ Dom(f)

3. f : RealSet → RealSet
4. y = f(x)
5. diff(x) ≠ 0
6. FunDeri(y, x, 1) = FunDeri(y, x, 1)
7. FunDeri(y, x, 2) = FunDeri(FunDeri(y, x, 1), x, 1)

GOAL:
FunDeri(y, x, 2) = frac(diff(x) * diff^{2}(y) - diff(y) * diff^{2}(x), diff(x)^{3})
