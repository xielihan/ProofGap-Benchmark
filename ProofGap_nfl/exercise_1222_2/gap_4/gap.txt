1. f : RealSet → RealSet
2. n ∈ NonNegIntegerSet ∧ n ∈ PosIntegerSet

4. forall (x), x ∈ RealSet ∧ -1 < x ∧ x < 1 ⇒ FunDeri(f, 1, 1)(x) = frac(2, sqrtn(2, 1 - x^{2})) * arcsin(x)

forall (x), x ∈ RealSet ∧ -1 < x ∧ x < 1 ⇒ (1 - x^{2}) * FunDeri(f, 1, 2)(x) - x * FunDeri(f, 1, 1)(x) - 2 = 0
