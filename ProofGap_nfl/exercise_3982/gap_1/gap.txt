PROOF GAP @1
ASSUM:
1. f : CartesianProd(RealSet, RealSet) → RealSet
2. u : CartesianProd(RealSet, RealSet) → RealSet
3. ContinuousFuncOn(f, CartesianProd(RealSet, RealSet))
4. ContinuousFuncOn(FunDeri(f, 2, 1), CartesianProd(RealSet, RealSet))
5. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ u(x, y) = frac(1, 2) * DefInt(0, x, (fun ξ [ξ ∈ RealSet] . DefInt(ξ - x + y, x + y - ξ, (fun η [η ∈ RealSet] . f(ξ, η)) * diff(fun η [η ∈ RealSet] . η))) * diff(fun ξ [ξ ∈ RealSet] . ξ))

GOAL:
forall (x), x ∈ RealSet ⇒ (forall (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ FunDeri(u, 1, 1)(x, y) = frac(1, 2) * DefInt(0, x, (fun ξ [ξ ∈ RealSet] . f(ξ, x + y - ξ) + f(ξ, ξ - x + y)) * diff(fun ξ [ξ ∈ RealSet] . ξ)))

METHOD:
[@method 根据 "含参变量积分求导公式" @]
