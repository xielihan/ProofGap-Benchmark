1. y : RealSet → RealSet
2. a ∈ RealSet
3. b ∈ RealSet

5. m ∈ RealSet

7. a + b * c^{2} = frac(m^{2}, c)
8. DiffableFuncAt(y, c)
GOAL:
FunDeri(fun x [x ∈ RealSet] . a + b * x^{2}, 1, 1)(c) = FunDeri(fun x [x ∈ RealSet] . frac(m^{2}, |x|), 1, 1)(c)
