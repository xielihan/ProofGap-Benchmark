ASSUM:
1. z : CartesianProd(RealSet, RealSet) → RealSet
2. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ z(x, y) = x^{3} + y^{3} - 3 * x * y
3. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ FunDeri(z, 1, 1)(x, y) = 3 * x^{2} - 3 * y
4. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ FunDeri(z, 2, 1)(x, y) = 3 * y^{2} - 3 * x

forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ (FunDeri(z, 1, 1)(x, y) = 0 ∧ FunDeri(z, 2, 1)(x, y) = 0 ⇔ (x, y) = (0, 0) ∨ (x, y) = (1, 1))
