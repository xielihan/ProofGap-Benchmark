PROOF GAP @1
ASSUM:
1. f : RealSet → RealSet
2. m : RealSet → RealSet
3. M : RealSet → RealSet
4. a ∈ RealSet
5. b ∈ RealSet
6. a < b
7. ContinuousFuncOn(f, [a, b])
8. forall (x), x ∈ RealSet ∧ x ∈ [a, b] ⇒ (forall (ξ), ξ ∈ RealSet ∧ ξ ∈ [a, x] ⇒ m(x) = inf({ f(ξ) | ξ ∈ RealSet, ξ ∈ [a, x] }))
9. forall (x), x ∈ RealSet ∧ x ∈ [a, b] ⇒ (forall (ξ), ξ ∈ RealSet ∧ ξ ∈ [a, x] ⇒ M(x) = sup({ f(ξ) | ξ ∈ RealSet, ξ ∈ [a, x] }))

GOAL:
ContinuousFuncOn(f, [a, b])
