ASSUM:
1. u : CartesianProd(RealSet, RealSet) → RealSet
2. ξ ∈ RealSet
3. η ∈ RealSet
4. x ∈ RealSet
5. y ∈ RealSet
6. D ⊆ CartesianProd(RealSet, RealSet) ∧ (ξ, η) ∈ D
7. f : CartesianProd(RealSet, RealSet) → RealSet ∧ FuncOfClassKOn(f, D, 2)
8. ξ = x + y
9. η = x - y
10. u(x, y) = f(ξ, η)
11. (x, y) ∈ RealSet

GOAL:
diff(ξ) = diff(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . x) + diff(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . y)
