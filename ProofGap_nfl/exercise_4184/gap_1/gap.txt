ASSUM:
1. a ∈ RealSet
2. p ∈ RealSet
3. φ : CartesianProd(RealSet, RealSet) → RealSet
4. D ⊆ CartesianProd(RealSet, RealSet)
5. T ⊆ CartesianProd(RealSet, RealSet)

exists (m) (M), m ∈ RealSet ∧ M ∈ RealSet ∧ (forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ∧ 0 ≤ x ∧ x ≤ a ∧ 0 ≤ y ∧ y ≤ a ∧ x ≠ y ⇒ frac(m, |x - y|^{p}) ≤ frac(|φ(x, y)|, |x - y|^{p}) ∧ frac(|φ(x, y)|, |x - y|^{p}) ≤ frac(M, |x - y|^{p}))
