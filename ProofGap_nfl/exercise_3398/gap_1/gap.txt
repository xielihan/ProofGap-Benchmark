1. D ⊆ CartesianProd(RealSet, RealSet)
2. F : D → RealSet
3. z : CartesianProd(RealSet, RealSet) → RealSet

5. FuncOfClassK(z, 2)

7. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ F(x * z(x, y), y * z(x, y)) = 0
