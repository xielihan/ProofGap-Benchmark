1. u : CartesianProd(RealSet, RealSet) → RealSet
2. v : CartesianProd(RealSet, RealSet) → RealSet
3. w : CartesianProd(RealSet, RealSet) → RealSet
4. z : CartesianProd(RealSet, RealSet) → RealSet
5. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ∧ |x| > |y| ⇒ u(x, y) = x + y ∧ v(x, y) = x - y ∧ w(x, y) = frac(z(x, y), sqrtn(2, x^{2} - y^{2}))
6. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ∧ |x| > |y| ⇒ FunDeri(z, x, 2)(x, y) + FunDeri(z, y, 2)(x, y) = 2 * frac(x * FunDeri(z, x, 1)(x, y) - y * FunDeri(z, y, 1)(x, y), x^{2} - y^{2}) - frac(3 * (x^{2} + y^{2}) * z(x, y), (x^{2} - y^{2})^{2})
7. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ∧ |x| > |y| ⇒ frac(1, x^{2} - y^{2}) * FunDeri(z, x, 2)(x, y) + frac(1, x^{2} - y^{2}) * FunDeri(z, y, 2)(x, y) - frac(2 * x, (x^{2} - y^{2})^{2}) * FunDeri(z, x, 1)(x, y) + frac(2 * y, (x^{2} - y^{2})^{2}) * FunDeri(z, y, 1)(x, y) = -frac(3 * (x^{2} + y^{2}) * z(x, y), (x^{2} - y^{2})^{3})
8. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ∧ |x| > |y| ⇒ frac(pdiff, pdiff(x))(frac(1, x^{2} - y^{2}) * FunDeri(z, x, 1), x, y) + frac(pdiff, pdiff(y))(frac(1, x^{2} - y^{2}) * FunDeri(z, y, 1), x, y) = -frac(3 * (x^{2} + y^{2}) * z(x, y), (x^{2} - y^{2})^{3})
