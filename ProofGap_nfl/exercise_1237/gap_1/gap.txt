PROOF GAP @1
ASSUM:
1. f : RealSet → RealSet
2. a ∈ RealSet
3. b ∈ RealSet
4. -∞ ≤ a
5. a < b
6. b ≤ +∞
7. forall (x), x ∈ RealSet ∧ x ∈ IntervalLoRo(a, b) ⇒ DiffableFuncAt(f, x)
8. forall (x), x ∈ RealSet ⇒ lim_{ x → a^+ } (f(x)) = lim_{ x → b^- } (f(x))
9. a > -∞ ⇒ b < +∞ ⇒ A = lim_{ x → a^+ } (f(x))
10. a > -∞ ⇒ b < +∞ ⇒ F = (fun x [x ∈ RealSet] . cases{ f(x) if x ∈ IntervalLoRo(a, b); A if x = a ∨ x = b })

GOAL:
a > -∞ ⇒ b < +∞ ⇒ ContinuousFuncOn(F, [a, b])
