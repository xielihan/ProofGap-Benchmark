1. a ∈ RealSet
2. f : RealSet → RealSet
3. L ∈ RealSet
4. x ∈ RealSet

7. lim_{ x → +∞ } (f(x)) = L

9. forall (ε), ε ∈ RealSet ∧ ε > 0 ⇒ (exists (X), X ∈ RealSet ∧ X > a ∧ ContinuousFuncOn(f, [a, X + 1]))
