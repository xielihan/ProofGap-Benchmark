1. p ∈ RealSet
2. p < 0 ⇒ lim_{ n → +∞ } (1 / n^{p}) = +∞

5. p = 0 ⇒ DivergentSeries(sum_{ n = 1 }^{ +∞ } ((-1)^{n - 1} / n^{p}))

forall (a), a : NonNegIntegerSet → RealSet ∧ 0 < p ∧ p ≤ 1 ∧ a = (fun n [n ∈ NonNegIntegerSet ∧ n ∈ PosIntegerSet] . 1 / n^{p}) ⇒ (forall (n), n ∈ NonNegIntegerSet ∧ n ∈ PosIntegerSet ⇒ a(n) > 0)
