6. ρ : CartesianProd(RealSet, RealSet) → RealSet

13. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ ρ(x, y) = |y|
14. M = ScalarCurveInt(C, ρ * diff(s))
