1. y : RealSet → RealSet

forall (x), x ∈ RealSet ⇒ (forall (k), k ∈ NonNegIntegerSet ∧ k ∈ PosIntegerSet ∧ 2 * k * π < frac(π, x) ∧ frac(π, x) < (2 * k + 1) * π ⇒ FunDeri(y, 1, 1)(x) > 0)
