ASSUM:
1. φ : CartesianProd(RealSet, RealSet) → RealSet
2. m ∈ RealSet ∧ m > 0

6. forall (x), x ∈ RealSet ⇒ (forall (y), y ∈ RealSet ⇒ D = { (x, y) | x ∈ RealSet ∧ y ∈ RealSet ∧ x^{2} + y^{2} > 1 })
7. Defined(φ, D)
8. forall (x), x ∈ RealSet ⇒ (forall (y), y ∈ RealSet ∧ (x, y) ∈ D ⇒ m ≤ |φ(x, y)| ∧ |φ(x, y)| ≤ M)
9. forall (x), x ∈ RealSet ⇒ (forall (y), y ∈ RealSet ∧ (x, y) ∈ D ⇒ frac(m, (x^{2} + y^{2})^{p}) ≤ |frac(φ(x, y), (x^{2} + y^{2})^{p})| ∧ |frac(φ(x, y), (x^{2} + y^{2})^{p})| ≤ frac(M, (x^{2} + y^{2})^{p}))
10. IntegrableFuncOn(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . frac(φ(x, y), (x^{2} + y^{2})^{p}), D) ⇔ IntegrableFuncOn(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . frac(1, (x^{2} + y^{2})^{p}), D)
11. forall (x), x ∈ RealSet ⇒ (forall (y), y ∈ RealSet ⇒ (forall (r), r ∈ RealSet ∧ r ≥ 1 ⇒ VolumeInt(D, frac(1, (x^{2} + y^{2})^{p}) * diff(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . x) * diff(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . y)) = DefInt(0, 2 * π, diff(fun theta [theta ∈ RealSet] . theta)) * DefInt(1, +∞, frac(r, r^{2 * p}) * diff(fun r [r ∈ RealSet ∧ r ≥ 1] . r))))
