PROOF GAP @10
ASSUM:
1. f : RealSet → RealSet
2. g : RealSet → RealSet
3. forall (x), x ∈ RealSet ⇒ f(x) = x^{2}
4. forall (x), x ∈ RealSet ⇒ g(x) = x^{3}
5. ContinuouslyDiffableFuncOn(f, [-1, 1])
6. ContinuouslyDiffableFuncOn(g, [-1, 1])
7. g(-1) ≠ g(1)
8. forall (x), x ∈ RealSet ∧ x = 0 ⇒ FunDeri(f, 1, 1)(x)^{2} + FunDeri(g, 1, 1)(x)^{2} = 4 * x^{2} + 9 * x^{4}
9. forall (x), x ∈ RealSet ∧ x = 0 ⇒ 4 * x^{2} + 9 * x^{4} = 0
10. forall (x), x ∈ RealSet ∧ x = 0 ⇒ FunDeri(f, 1, 1)(x)^{2} + FunDeri(g, 1, 1)(x)^{2} = 0
11. FunDeri(f, 1, 1)(0) = 0
12. FunDeri(g, 1, 1)(0) = 0
13. frac(f(1) - f(-1), g(1) - g(-1)) = 0

GOAL:
forall (ξ), ξ ∈ RealSet ∧ ξ ∈ IntervalLoRo(-1, 1) ∧ ξ ≠ 0 ⇒ frac(FunDeri(f, 1, 1)(ξ), FunDeri(g, 1, 1)(ξ)) = frac(2 * ξ, 3 * ξ^{2})
