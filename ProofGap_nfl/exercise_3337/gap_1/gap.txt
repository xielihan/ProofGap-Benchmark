ASSUM:
1. z : CartesianProd(RealSet, RealSet) → RealSet
2. φ : RealSet → RealSet
3. ψ : RealSet → RealSet
4. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ z(x, y) = φ(x) * ψ(y)
5. DiffableFunc(fun x [x ∈ RealSet] . φ(x))
6. DiffableFunc(fun y [y ∈ RealSet] . ψ(y))

forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ FunDeri(fun x, y [x ∈ RealSet ∧ y ∈ RealSet] . z(x, y), 1, 1)(x, y) = FunDeri(fun x [x ∈ RealSet] . φ(x), 1, 1)(x) * ψ(y)
