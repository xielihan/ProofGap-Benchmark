PROOF GAP @3
ASSUM:
1. F : CartesianProd(NonNegIntegerSet, RealSet) → RealSet
2. f : RealSet → RealSet
3. forall (n) (x), n ∈ NonNegIntegerSet ∧ x ∈ RealSet ∧ n ∈ PosIntegerSet ∧ x ∈ [0, 2] ⇒ F(n, x) = sqrtn(n, 1 + x^{n})
4. forall (x), x ∈ RealSet ∧ x ∈ [0, 2] ⇒ f(x) = cases{ 1 if 0 ≤ x ∧ x ≤ 1; x if 1 < x ∧ x ≤ 2 }
5. forall (n) (x), n ∈ NonNegIntegerSet ∧ x ∈ RealSet ∧ n ∈ PosIntegerSet ∧ 0 ≤ x ∧ x ≤ 1 ⇒ |F(n, x) - f(x)| = |sqrtn(n, 1 + x^{n}) - 1|
6. forall (n) (x), n ∈ NonNegIntegerSet ∧ x ∈ RealSet ∧ n ∈ PosIntegerSet ∧ 0 ≤ x ∧ x ≤ 1 ⇒ |F(n, x) - f(x)| = frac(x^{n}, sum_{ k = 0 }^{ n - 1 } ((1 + x^{n})^{frac(k, n)}))

GOAL:
forall (n) (x), n ∈ NonNegIntegerSet ∧ x ∈ RealSet ∧ n ∈ PosIntegerSet ∧ 0 ≤ x ∧ x ≤ 1 ⇒ frac(x^{n}, sum_{ k = 0 }^{ n - 1 } ((1 + x^{n})^{frac(k, n)})) < frac(1, n)
