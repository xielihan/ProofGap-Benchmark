1. y : RealSet → RealSet

forall (x), x ∈ RealSet ⇒ FunDeri(y, 1, 2)(x) = FunDeri(fun x [x ∈ RealSet] . frac(x, sqrtn(2, 1 + x^{2})), 1, 1)(x)
