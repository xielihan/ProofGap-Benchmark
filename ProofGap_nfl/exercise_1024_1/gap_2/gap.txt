PROOF GAP @2
ASSUM:
1. P : CartesianProd(NonNegIntegerSet, RealSet) → RealSet
2. `P̅` : CartesianProd(NonNegIntegerSet, RealSet) → RealSet
3. forall (k) (n) (x), k ∈ NonNegIntegerSet ∧ n ∈ NonNegIntegerSet ∧ n ∈ PosIntegerSet ∧ x ∈ RealSet ⇒ P(n, x) = sum_{ k = 1 }^{ n } (k * x^{k - 1})
4. forall (k) (n), k ∈ NonNegIntegerSet ∧ n ∈ NonNegIntegerSet ∧ n ∈ PosIntegerSet ⇒ `P̅`(n) = (fun x [x ∈ RealSet] . sum_{ k = 1 }^{ n } (x^{k}))
5. forall (n) (x), n ∈ NonNegIntegerSet ∧ n ∈ PosIntegerSet ∧ x ∈ RealSet ∧ x ≠ 1 ⇒ `P̅`(n, x) = frac(x * (1 - x^{n}), 1 - x)

GOAL:
forall (n) (x), n ∈ NonNegIntegerSet ∧ n ∈ PosIntegerSet ∧ x ∈ RealSet ⇒ FunDeri(`P̅`(n), 1, 1)(x) = P(n, x)
