1. f : RealSet → RealSet
2. a ∈ RealSet
3. b ∈ RealSet
4. a < b

7. |f(b)| ≤ 1
8. forall (y), y ∈ RealSet ∧ y ∈ [-1, 1] ⇒ (exists (x), x ∈ RealSet ∧ x ∈ (a, b] ∧ f(x) = y)
