PROOF GAP @2
ASSUM:
1. f : RealSet → RealSet
2. F : RealSet → RealSet
3. l ∈ RealSet ∧ l > 0
4. Defined(f, [-l, l])
5. ContinuousFuncOn(f, [-l, l])
6. forall (x), x ∈ RealSet ∧ x ∈ [-l, l] ⇒ FunDeri(F, 1, 1)(x) = f(x)
7. EvenFunc(f) ⇒ (forall (x), x ∈ RealSet ∧ x ∈ [-l, l] ⇒ f(-x) = f(x))

GOAL:
EvenFunc(f) ⇒ (forall (x), x ∈ RealSet ∧ x ∈ [-l, l] ⇒ f(x) = FunDeri(F, 1, 1)(x) ∧ f(-x) = -FunDeri(F, 1, 1)(-x))
