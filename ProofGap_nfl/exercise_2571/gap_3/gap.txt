ASSUM:
1. a : NonNegIntegerSet → RealSet ∧ IsSeq(a)
2. α : NonNegIntegerSet → RealSet
3. PosSeq(a)
4. MonoDecSeq(a)
5. forall (n), n ∈ NonNegIntegerSet ∧ n ∈ PosIntegerSet ⇒ ConvergentSeries(sum_{ n = 1 }^{ +∞ } (a(n)))
6. forall (m), m ∈ NonNegIntegerSet ⇒ (forall (k), k ∈ NonNegIntegerSet ∧ m ∈ PosIntegerSet ∧ k ∈ PosIntegerSet ⇒ α(m) = sum_{ k = m + 1 }^{ +∞ } (a(k)))
7. forall (m), m ∈ NonNegIntegerSet ⇒ (forall (n), n ∈ NonNegIntegerSet ∧ m ∈ PosIntegerSet ∧ n ∈ PosIntegerSet ∧ n > m ⇒ (n - m) * a(n) ≤ sum_{ k = m + 1 }^{ n } (a(k)))
8. forall (m), m ∈ NonNegIntegerSet ⇒ (forall (n), n ∈ NonNegIntegerSet ∧ m ∈ PosIntegerSet ∧ n ∈ PosIntegerSet ∧ n > m ⇒ sum_{ k = m + 1 }^{ n } (a(k)) < α(m))

forall (m), m ∈ NonNegIntegerSet ⇒ (forall (n), n ∈ NonNegIntegerSet ∧ m ∈ PosIntegerSet ∧ n ∈ PosIntegerSet ∧ n > m ⇒ n * a(n) < frac(n, n - m) * α(m))
