5. forall (x), x ∈ RealSet ∧ x ∈ [-1, 1] ⇒ y(x) = x^{2}

forall (x), x ∈ RealSet ∧ x ∈ [-1, 1] ⇒ diff(y) = 2 * x * diff(fun x [x ∈ RealSet ∧ x ∈ [-1, 1]] . x)
