1. y : RealSet → RealSet

4. forall (x), x ∈ RealSet ∧ x ≠ 1 ⇒ (FunDeri(y, 1, 1)(x) = 0 ⇔ x = frac(3, 4))

GOAL:
forall (x), x ∈ RealSet ∧ x < frac(3, 4) ⇒ FunDeri(y, 1, 1)(x) < 0
