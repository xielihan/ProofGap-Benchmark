PROOF GAP @1
ASSUM:
1. x : NonNegIntegerSet → RealSet
2. p : NonNegIntegerSet → NonNegIntegerSet
3. a ∈ RealSet
4. IsSeq(x)
5. forall (n), n ∈ NonNegIntegerSet ∧ n ∈ PosIntegerSet ⇒ x(n) ∈ RealSet
6. MonoIncSeq(x)
7. IsSubseqIndexFunc(p)
8. ConvergentSeqTo(x ∘ p, a)

GOAL:
forall (ε), ε ∈ RealSet ∧ ε > 0 ⇒ (exists (N), N ∈ NonNegIntegerSet ∧ N ∈ PosIntegerSet ∧ (forall (k), k ∈ NonNegIntegerSet ∧ k ∈ PosIntegerSet ∧ k > N ⇒ |x(p(k)) - a| < ε))
