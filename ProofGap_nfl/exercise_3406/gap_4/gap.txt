1. x : RealSet → RealSet
2. y : RealSet → RealSet
3. z : RealSet → RealSet
4. forall (t), t ∈ RealSet ∧ t ≠ 0 ⇒ x(t) = t + t^{-1}
5. forall (t), t ∈ RealSet ∧ t ≠ 0 ⇒ y(t) = t^{2} + t^{-2}
6. forall (t), t ∈ RealSet ∧ t ≠ 0 ⇒ z(t) = t^{3} + t^{-3}

9. forall (t), t ∈ RealSet ∧ t ≠ 0 ∧ t ≠ 1 ∧ t ≠ -1 ⇒ FunDeri(z, 1, 1)(t) = 3 * t^{2} - frac(3, t^{4})

forall (t), t ∈ RealSet ∧ t ≠ 0 ∧ t ≠ 1 ∧ t ≠ -1 ⇒ FunDeri(y, x, 1) = frac(FunDeri(y, 1, 1)(t), FunDeri(x, 1, 1)(t))
