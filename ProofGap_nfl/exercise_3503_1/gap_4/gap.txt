2. f : RealSet → RealSet
3. r : CartesianProd(RealSet, RealSet) → RealSet
4. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ r(x, y) = sqrtn(2, x^{2} + y^{2}) ∧ u(x, y) = f(r(x, y))
5. DiffableFunc(f)
6. DiffableFunc(FunDeri(f, 1, 1))
7. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ∧ x^{2} + y^{2} > 0 ⇒ FunDeri(u, 1, 2)(x, y) + FunDeri(u, 2, 2)(x, y) = 0
8. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ∧ x^{2} + y^{2} > 0 ⇒ FunDeri(u, 1, 1)(x, y) = FunDeri(f, 1, 1)(r(x, y)) * frac(x, r(x, y))
9. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ∧ x^{2} + y^{2} > 0 ⇒ FunDeri(u, 2, 1)(x, y) = FunDeri(f, 1, 1)(r(x, y)) * frac(y, r(x, y))
10. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ∧ x^{2} + y^{2} > 0 ⇒ FunDeri(u, 1, 2)(x, y) = frac(x^{2}, r(x, y)^{2}) * FunDeri(f, 1, 2)(r(x, y)) + frac(y^{2}, r(x, y)^{3}) * FunDeri(f, 1, 1)(r(x, y))
