ASSUM:
1. u : CartesianProd(RealSet, RealSet) → RealSet
2. v : CartesianProd(RealSet, RealSet) → RealSet
3. w : CartesianProd(RealSet, RealSet) → RealSet
4. z : CartesianProd(RealSet, RealSet) → RealSet
5. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ u(x, y) = y * z(x, y) - x ∧ v(x, y) = x * z(x, y) - y ∧ w(u(x, y), v(x, y)) = x * y - z(x, y)
6. forall (x) (y), x ∈ RealSet ∧ y ∈ RealSet ⇒ (x * y + z(x, y)) * FunDeri(z, 1, 1)(x, y) + (1 - y^{2}) * FunDeri(z, 2, 1)(x, y) = x + y * z(x, y)
