forall (p), p ∈ IntegerSet ⇒ (forall (q), q ∈ IntegerSet ∧ m ∈ NegIntegerSet ∧ n ∈ NegIntegerSet ∧ p = -m ∧ q = -n ⇒ p ∈ PosIntegerSet)
